Sub GO_CISCO()
    Dim thisB As Workbook
    Set thisB = ActiveWorkbook
    With Application.FileDialog(msoFileDialogFilePicker)
           .InitialFileName = thisB.Path
           .AllowMultiSelect = False
           .ButtonName = "OK"
           .Filters.Clear
           .Filters.Add Description:="Файлы Microsoft Excel", _
                Extensions:="*.xls; *.xlsx; *.xlsm"
           If .Show = 0 Then
                Exit Sub
           End If
           Set sB = Workbooks.Open(.SelectedItems(1))
    End With
    Select Case mSh.[B3]
    Case "KZ"
        Call Macros_Cisco_KZ
    Case "UZ", "TM", "AZ", "TJ"
    If mSh.CheckBoxes(2).Value < 0 Then
        Call Macros_Cisco_UZnew
        Else
        Call Macros_Cisco_shUZddp
        End If
        
    Case "GR", "AM"
        Call Macros_Cisco_GR
    ' Case "AM"
  '   Call Macros_Cisco_AM
        
    End Select
End Sub

Private Sub Create_File_Excel()
Dim Sh As Worksheet
Dim Shablon1 As Worksheet
    SaveFile
    Set Shablon1 = Macros.Worksheets("LenovoNEW")
    Set Rezult = Workbooks.Add
    Shablon1.Visible = True
    Shablon1.Activate
    Shablon1.Copy After:=Rezult.Sheets(Rezult.Sheets.count)
    Set Rez1 = ActiveSheet
    Rez1.Name = "Commercial Offer"
    Shablon1.Visible = False
    
    For Each Sh In Rezult.Worksheets
            If (Sh.Name <> "Commercial Offer") Then
                    Sh.Delete
            End If
    Next
    'Rezult.SaveAs FullName2
    Rezult.SaveAs FileName:= _
      Path & "\" & SpecName & ".xlsx", FileFormat:= _
xlOpenXMLWorkbook, CreateBackup:=False
End Sub
